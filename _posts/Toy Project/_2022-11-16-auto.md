def pre_process(result, keyword):
    address_type = [
        'SITEWHLADDR', 'ORG_RDNMA', 'ADDR', 'ADDRESS'
    ]
    year_type = [
        'APVPERMYMD', 'FOND_YMD', 'FOAS_MEMRD'
    ]
    tel_type = [
        'TEL', 'PHONE_NUM'
    ]

    if  result != '':
        if keyword in year_type:
            result = result[:4]
            return result
        elif keyword in address_type:
            try:
                result = result.split()[2]
            except:
                result = result.split()[1]
            return result
        elif keyword == 'KINDERNAME':
            n = result.find('유치원')
            result = result[:n]
            return result             
        elif keyword == 'TITLE':
            result = re.findall('[가-힣]+', result)[0]
            return result
        elif keyword in tel_type:
            result = result[-2:]
            return result
    else:
        return result

def stop(result, keyword):
    bool = False
    no_cond_col = [
        'APVPERMYMD', 'TITLE', 'FOND_YMD', 'FOAS_MEMRD', 'TEL', 'PHONE_NUM'
    ]
    address_type = [
        'ADDR', 'ADDRESS'
    ]

    if keyword in no_cond_col:
        if result != '':
            bool = True
    elif keyword == 'SITEWHLADDR':
        if '동' in result:
            bool = True           
    elif keyword == 'KINDERNAME':
        if len(result) == 2:
            bool = True     
    elif keyword == 'ORG_RDNMA':
        if not result.isdigit():
            bool = True         
    elif keyword in address_type:
        for loc in ['시','동','로','면','읍']:
            if loc in result:
                bool = True
                return bool
        if '로' in result:
            if not result.isdigit():
                bool = True
    return bool
